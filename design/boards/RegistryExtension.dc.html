Add it

One line: the newest 0.8.x
doze.hcl
module postgres "postgres" {
  version    = 18
  extensions = ["vector"]
}
Pinned to an exact version
doze.hcl
module postgres "postgres" {
  version = 18
  extension "vector" { version = "0.8.7" }
}

doze fetches the build for your Postgres major and platform once, checks its sha256, pins it in doze.lock and runs CREATE EXTENSION vector. Nothing to install, the same on a Mac and on Linux. A newer pin later runs ALTER EXTENSION vector UPDATE for you.

Use it from your code

CREATE EXTENSION is done for you. Make a vector column, then order by distance: <-> is L2, <=> cosine, <#> inner product.

SQL, in your migration
create table items (id bigserial primary key, embedding vector(3));
create index on items using hnsw (embedding vector_l2_ops);

select id from items order by embedding <-> '[3,1,2]' limit 5;
Python · pgvector + psycopg
from pgvector.psycopg import register_vector
import numpy as np, os, psycopg

conn = psycopg.connect(os.environ["DATABASE_URL"])
register_vector(conn)
conn.execute(
  "select id from items order by embedding <-> %s limit 5",
  (np.array([3, 1, 2]),))
Node · pgvector + pg
import pg from "pg";
import pgvector from "pgvector/pg";

const client = new pg.Client(process.env.DATABASE_URL);
await client.connect();
await pgvector.registerTypes(client);
await client.query(
  "select id from items order by embedding <-> $1 limit 5",
  [pgvector.toSql([3, 1, 2])]);
Go · pgvector-go + pgx
import (
  "github.com/pgvector/pgvector-go"
  pgxvec "github.com/pgvector/pgvector-go/pgx"
)

pgxvec.RegisterTypes(ctx, conn)
rows, err := conn.Query(ctx,
  "select id from items order by embedding <-> $1 limit 5",
  pgvector.NewVector([]float32{3, 1, 2}))

What doze.lock records

A bundle is pinned like a Postgres build: version and one sha256 per platform, written the first time each platform fetches it.

engines:
    pgext-vector-pg18:
        "0.8":
            resolved: 0.8.7
            source: mirror
            hashes:
                aarch64-apple-darwin: sha256:…

… is each platform's sha256, written on its first fetch. The module pin above it moves to postgres 0.3; doze modules upgrade moves both on purpose.
